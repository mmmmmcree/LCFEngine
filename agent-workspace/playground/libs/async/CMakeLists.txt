cmake_minimum_required(VERSION 3.28)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LCFAsync VERSION 0.1.0 LANGUAGES CXX)
    set_property(GLOBAL PROPERTY USE_FOLDERS ON)
    if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR AND NOT EXISTS "${CMAKE_BINARY_DIR}/.gitignore")
        file(WRITE "${CMAKE_BINARY_DIR}/.gitignore" "*\n")
    endif()
    enable_testing()
endif()
include(cmake/options.cmake)
if(LCF_ASYNC_ENABLE_ASIO)
    add_subdirectory(asio)
endif()
if(LCF_ASYNC_ENABLE_STDEXEC)
    add_subdirectory(stdexec)
endif()
if(LCF_ASYNC_BUILD_TESTS)
    add_subdirectory(tests)
endif()
if(LCF_ASYNC_INSTALL)
    include(cmake/install.cmake)
endif()
