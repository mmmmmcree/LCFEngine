cmake_minimum_required(VERSION 3.28)
project(AsyncConsumer LANGUAGES CXX)

set_property(GLOBAL PROPERTY USE_FOLDERS ON)
if(NOT CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR AND NOT EXISTS "${CMAKE_BINARY_DIR}/.gitignore")
    file(WRITE "${CMAKE_BINARY_DIR}/.gitignore" "*\n")
endif()
set(LCF_ASYNC_CONSUMPTION "SUBDIRECTORY" CACHE STRING "Async source: SUBDIRECTORY, FETCH, or PACKAGE")
set_property(CACHE LCF_ASYNC_CONSUMPTION PROPERTY STRINGS SUBDIRECTORY FETCH PACKAGE)
set(LCF_ASYNC_CONSUMER_BACKEND "asio" CACHE STRING "Async implementation: asio or stdexec")
set_property(CACHE LCF_ASYNC_CONSUMER_BACKEND PROPERTY STRINGS asio stdexec)
if(NOT LCF_ASYNC_CONSUMER_BACKEND MATCHES "^(asio|stdexec)$")
    message(FATAL_ERROR "Unknown async implementation: ${LCF_ASYNC_CONSUMER_BACKEND}")
endif()

if(LCF_ASYNC_CONSUMPTION STREQUAL "PACKAGE")
    find_package(LCFAsync CONFIG REQUIRED COMPONENTS ${LCF_ASYNC_CONSUMER_BACKEND})
else()
    set(LCF_ASYNC_BUILD_TESTS OFF)
    set(LCF_ASYNC_INSTALL OFF)
    set(LCF_ASYNC_ENABLE_ASIO OFF)
    set(LCF_ASYNC_ENABLE_STDEXEC OFF)
    string(TOUPPER "${LCF_ASYNC_CONSUMER_BACKEND}" async_backend)
    set(LCF_ASYNC_ENABLE_${async_backend} ON)
    if(LCF_ASYNC_CONSUMPTION STREQUAL "SUBDIRECTORY")
        set(LCF_ASYNC_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../async" CACHE PATH "Async submodule or source checkout")
        if(NOT EXISTS "${LCF_ASYNC_SOURCE_DIR}/CMakeLists.txt")
            message(FATAL_ERROR "Initialize the async submodule, or set LCF_ASYNC_SOURCE_DIR")
        endif()
        add_subdirectory("${LCF_ASYNC_SOURCE_DIR}" async-build)
    elseif(LCF_ASYNC_CONSUMPTION STREQUAL "FETCH")
        set(LCF_ASYNC_GIT_REPOSITORY "" CACHE STRING "Async repository URL or local bare repository")
        set(LCF_ASYNC_GIT_TAG "" CACHE STRING "Pinned async commit")
        if(NOT LCF_ASYNC_GIT_REPOSITORY OR NOT LCF_ASYNC_GIT_TAG)
            message(FATAL_ERROR "FETCH requires LCF_ASYNC_GIT_REPOSITORY and LCF_ASYNC_GIT_TAG")
        endif()
        include(FetchContent)
        string(REPLACE "\\" "/" async_git_repository "${LCF_ASYNC_GIT_REPOSITORY}")
        FetchContent_Declare(lcf_async GIT_REPOSITORY "${async_git_repository}" GIT_TAG "${LCF_ASYNC_GIT_TAG}")
        FetchContent_MakeAvailable(lcf_async)
    else()
        message(FATAL_ERROR "Unknown async consumption mode: ${LCF_ASYNC_CONSUMPTION}")
    endif()
endif()

add_executable(async_consumer main.cpp)
target_link_libraries(async_consumer PRIVATE lcf::async_${LCF_ASYNC_CONSUMER_BACKEND})
if(LCF_ASYNC_CONSUMER_BACKEND STREQUAL "stdexec")
    target_compile_definitions(async_consumer PRIVATE LCF_EXAMPLE_STDEXEC=1)
endif()
set_target_properties(async_consumer PROPERTIES CXX_EXTENSIONS OFF FOLDER "examples/async/consumer")
